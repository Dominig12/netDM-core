cmake_minimum_required(VERSION 3.21)
project(netdm_byond LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4 OR (NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux"))
  message(FATAL_ERROR "BYOND bridge requires a 32-bit Windows or Linux toolchain.")
endif()
find_package(Threads REQUIRED)
add_library(netdm_byond SHARED netdm_bridge.cpp)
target_compile_features(netdm_byond PRIVATE cxx_std_17)
target_link_libraries(netdm_byond PRIVATE Threads::Threads)
set_target_properties(netdm_byond PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN YES)
if(WIN32)
  target_sources(netdm_byond PRIVATE netdm_bridge.def)
  target_compile_definitions(netdm_byond PRIVATE _WIN32_WINNT=0x0601 UNICODE _UNICODE)
  add_executable(netdm_bridge_test test_bridge.cpp)
  target_compile_features(netdm_bridge_test PRIVATE cxx_std_17)
  add_dependencies(netdm_bridge_test netdm_byond)
  if(MINGW)
    # DreamDaemon must load the DLL without MinGW runtime DLLs next to it: link the
    # compiler runtime, winpthreads and libstdc++ statically in both binaries.
    target_link_options(netdm_byond PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(netdm_bridge_test PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
