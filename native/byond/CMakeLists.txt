cmake_minimum_required(VERSION 3.21)
project(netdm_byond LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "BYOND bridge must compile as Windows Win32 (x86).")
endif()
add_library(netdm_byond SHARED netdm_bridge.cpp netdm_bridge.def)
target_compile_features(netdm_byond PRIVATE cxx_std_17)
target_compile_definitions(netdm_byond PRIVATE UNICODE _UNICODE)
set_target_properties(netdm_byond PROPERTIES OUTPUT_NAME "netdm_byond")
add_executable(netdm_bridge_test test_bridge.cpp)
target_compile_features(netdm_bridge_test PRIVATE cxx_std_17)
add_dependencies(netdm_bridge_test netdm_byond)
